cmake_minimum_required(VERSION 3.20)
project(http_sdk_contracts LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(CURL 8.11.0 EXACT REQUIRED)
if(NOT DEFINED CPR_SOURCE_DIR)
  message(FATAL_ERROR "CPR_SOURCE_DIR must identify the pinned CPR 1.11.1 source")
endif()
set(CPR_USE_SYSTEM_CURL ON CACHE BOOL "Use the pinned local curl build" FORCE)
set(CPR_ENABLE_SSL OFF CACHE BOOL "This fixture exercises loopback HTTP only" FORCE)
set(CPR_BUILD_TESTS OFF CACHE BOOL "Do not build upstream tests" FORCE)
add_subdirectory("${CPR_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/cpr")
add_executable(curl_c_contracts curl_contracts.c ../../c/clients.c)
target_link_libraries(curl_c_contracts PRIVATE CURL::libcurl)
add_executable(curl_cpp_contracts curl_cpp_contracts.cpp ../../cpp/userdata.cpp)
target_link_libraries(curl_cpp_contracts PRIVATE CURL::libcurl)
add_executable(cpr_contracts cpr_contracts.cpp ../../cpp/precision.cpp)
target_link_libraries(cpr_contracts PRIVATE cpr::cpr)
